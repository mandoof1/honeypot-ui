# The honeypot engine is the internet-facing component; its dependency
# surface is kept deliberately minimal. httpx is used to ship captured
# sessions to the backend, cryptography to serve the HTTPS emulator.
httpx==0.28.1
# asyncssh 2.24 requires cryptography>=48.0.1. This was pinned at 44.0.0
# alongside it, which made the set unresolvable: `pip install -r` failed
# and the engine image could not be built.
cryptography==50.0.1

# Real SSH transport for the SSH emulator. There is no way to speak the
# protocol without one, and hand-rolling key exchange on the internet-facing
# component would be a far worse trade than a maintained library.
asyncssh==2.24.0
